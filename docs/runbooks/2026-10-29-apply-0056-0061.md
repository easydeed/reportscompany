## Step 1 — read-only checks (change nothing)

```sql
-- A. Where the database is. Expect 0055_affiliate_branding_postal_address.sql as the
--    newest numbered row, and seed_demo_account.sql recorded. Nothing from 0056 on.
SELECT filename, applied_at FROM schema_migrations ORDER BY filename DESC LIMIT 4;

-- B. Nothing from 0060 is half there. Expect: seat_table empty, rep_seats_column 0.
SELECT to_regclass('public.rep_seat_assignments') AS seat_table,
       (SELECT count(*) FROM information_schema.columns
         WHERE table_name = 'accounts' AND column_name = 'rep_seats') AS rep_seats_column;

-- C. 0056 will do nothing. Expect one row for 'free'.
SELECT plan_slug, monthly_report_limit, market_reports_limit, schedules_limit,
       property_reports_per_month FROM plans WHERE plan_slug = 'free';

-- D. 0057 CHANGES ROWS: sets html_url to NULL on this many market reports.
SELECT count(*) AS print_links_to_clear FROM report_generations WHERE html_url LIKE '%/print/%';

-- E. 0058 CHANGES ROWS: every account on theme 1 or 4 moves to 5. NULL stays NULL.
SELECT default_theme_id, count(*) FROM accounts GROUP BY 1 ORDER BY 1;

-- F. 0059 CHANGES ROWS: clears the accent on accounts still on BOTH colour defaults.
SELECT count(*) AS accents_to_clear FROM accounts
 WHERE lower(secondary_color) = '#ffffff' AND lower(primary_color) = '#03374f';
SELECT count(*) AS white_accent_left_alone FROM accounts
 WHERE lower(secondary_color) = '#ffffff' AND lower(primary_color) <> '#03374f';

-- G. 0060 ADDS ROWS: one open seat per rep under a company whose owner user is
--    active. Expect Pacific Coast Title = 8.
SELECT c.name AS company, count(*) AS seats_to_open
  FROM accounts a JOIN accounts c ON c.id = a.parent_account_id
 WHERE a.account_type = 'INDUSTRY_AFFILIATE'
   AND NOT EXISTS (SELECT 1 FROM account_users au JOIN users u ON u.id = au.user_id
                    WHERE au.account_id = a.id AND au.role = 'OWNER' AND u.is_active = FALSE)
 GROUP BY c.name ORDER BY c.name;

-- H. Pacific Coast's reps, one row each. Expect 8 with gets_a_seat = t.
SELECT a.name AS rep, u.email, u.is_active AS user_active, a.is_active AS account_active,
       (u.email_verified AND u.password_hash IS NOT NULL) AS joined,
       COALESCE(u.is_active, TRUE) AS gets_a_seat
  FROM accounts a JOIN accounts c ON c.id = a.parent_account_id
  LEFT JOIN account_users au ON au.account_id = a.id AND au.role = 'OWNER'
  LEFT JOIN users u ON u.id = au.user_id
 WHERE a.account_type = 'INDUSTRY_AFFILIATE' AND c.name ILIKE 'Pacific Coast%'
 ORDER BY a.name;
```
